Primero, creamos una base de datos de ejemplo. La función set.seed() asegura que siempre obtengamos los mismos resultados “aleatorios”.

set.seed(123) # Para que los resultados aleatorios sean reproducibles

encuesta_ejemplo <- tibble(
  folio = 1:10,
  edad = sample(18:70, 10, replace = TRUE),
  sexo = sample(c("Hombre", "Mujer"), 10, replace = TRUE),
  region = sample(c("Metropolitana", "Valparaíso", "Biobío", "Maule"), 10, replace = TRUE),
  educ = sample(c("Básica", "Media", "Superior"), 10, replace = TRUE),
  ingreso = round(runif(10, 350000, 1500000), 0)
)

print(encuesta_ejemplo)
